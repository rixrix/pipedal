set(PIPEDAL_PITCH_JUCE_SOURCE "" CACHE PATH "Existing JUCE 9.0.3 checkout (never downloaded)")
set(PIPEDAL_PITCH_REFERENCE_SOURCE "" CACHE PATH "Pinned TONE3000 v0.0.12 checkout for independent DSP tests")
if(NOT EXISTS "${PIPEDAL_PITCH_JUCE_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Set PIPEDAL_PITCH_JUCE_SOURCE to an existing JUCE 9.0.3 checkout")
endif()
file(READ "${PIPEDAL_PITCH_JUCE_SOURCE}/CMakeLists.txt" juce_project)
if(NOT juce_project MATCHES "project\\(JUCE VERSION 9\\.0\\.3 ")
    message(FATAL_ERROR "This pitch bundle requires pinned JUCE 9.0.3")
endif()
function(check_pitch_hash path expected)
    file(SHA256 "${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Pinned pitch source hash differs: ${path}")
    endif()
endfunction()
set(header_hash 6dfbd00a6c74b0020ba72e6a5f58c0b98d2c98f4c7975a7fcb9d1be6c36d0c30)
set(source_hash 069b6c62b96063c04d8fa574c7c8e5e249801acc65fb5e600eefe96194f63e3a)
check_pitch_hash("${CMAKE_CURRENT_SOURCE_DIR}/vendor/PitchShift.h" "${header_hash}")
check_pitch_hash("${CMAKE_CURRENT_SOURCE_DIR}/vendor/PitchShift.cpp" "${source_hash}")
set(JUCE_MODULES_ONLY ON CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PIPEDAL_PITCH_JUCE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
set(lv2_sdk "${PIPEDAL_PITCH_JUCE_SOURCE}/modules/juce_audio_processors_headless/format_types/LV2_SDK/lv2")
if(NOT EXISTS "${lv2_sdk}/lv2/core/lv2.h")
    message(FATAL_ERROR "JUCE's pinned LV2 SDK headers are missing")
endif()
function(configure_pitch_target target)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${lv2_sdk}" "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
    target_compile_definitions(${target} PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1
        JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0
        JUCE_USE_MP3AUDIOFORMAT=0 JUCE_USE_LAME_AUDIO_FORMAT=0 JUCE_USE_OPUS=0)
    target_link_libraries(${target} PRIVATE juce::juce_dsp)
    set_property(TARGET ${target} PROPERTY POSITION_INDEPENDENT_CODE ON)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target} PRIVATE -fno-fast-math -fvisibility=hidden)
    endif()
endfunction()
add_library(t3k_pitch MODULE T3KPitch.cpp vendor/PitchShift.cpp)
configure_pitch_target(t3k_pitch)
set(bundle "${CMAKE_CURRENT_BINARY_DIR}/T3KPitch.lv2")
set_target_properties(t3k_pitch PROPERTIES PREFIX "" SUFFIX ".so" LIBRARY_OUTPUT_DIRECTORY "${bundle}")
file(MAKE_DIRECTORY "${bundle}")
foreach(file manifest.ttl pitch.ttl NOTICE vendor/TONE3000-LICENSE vendor/JUCE-LICENSE.md)
    get_filename_component(name "${file}" NAME)
    configure_file("${file}" "${bundle}/${name}" COPYONLY)
endforeach()
install(TARGETS t3k_pitch LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/lv2/T3KPitch.lv2")
install(FILES manifest.ttl pitch.ttl NOTICE vendor/TONE3000-LICENSE vendor/JUCE-LICENSE.md
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/lv2/T3KPitch.lv2")
